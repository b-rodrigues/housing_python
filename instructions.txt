# Get python installed by any means on your computer
# you could install python using pyenv:
pyenv install 3.12.1

# switch to project directory and set up global python version
pyenv global 3.12.1

# Install pipenv
pip install pipenv --user

# install packages in your project's directory
pipenv install plotnine polars pandas beautifulsoup4

# how to install a package from github using pipenv  
pipenv install -e git+https://github.com/has2k1/plotnine.git@main#egg=plotnine


### Much better ###

# Install micromamba

# Install a python interpreter and pipenv

# This will create an environment called main with pipenv and the required python for the project
micromamba create -n main pipenv python=3.12.1

# Activate this environment
micromamba activate main

# you can now use pipenv to install packages and a project-specific python interpreter
cd ~/myproject
pipenv install polars plotnine

# In the lockfile, you can adjust the python version by adding the .patch to the version number then run "pipenv update"
# to make sure that this is reflected in the lockfile as well

# if you need another python version for another project, install the interpreter using micromamba in a 
# new environment
 micromamba create -n 3.10.12 pipenv python=3.10.12

# activate the environment, and use it to install packages using pipenv and recreate the Pipfiles
# instead of activating the environment, you could do
micromamba run -n 3.10.12 pipenv install polars==1.0.0 plotnine

# drop into the shell using pipenv shell (if you didn’t activate the micromamba shell before, use micromamba run -n 3.10.12 pipenv shell)
